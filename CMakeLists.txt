cmake_minimum_required(VERSION 3.20)
project(prento VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(PRENTO_SANITIZERS "Build with address/undefined sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(PRENTO_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(prento src/main.cpp src/collector.cpp src/producer.cpp src/sink.cpp)
target_include_directories(prento PRIVATE include)
target_link_libraries(prento PRIVATE Threads::Threads)
enable_testing()
add_executable(core-test tests/core_test.cpp)
target_include_directories(core-test PRIVATE include)
add_test(NAME core COMMAND core-test)
install(TARGETS prento RUNTIME DESTINATION bin)
